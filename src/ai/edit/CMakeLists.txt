# wabcut_ai_edit — transcript-first editing (AI-3, docs/ai.md "Text-based editing"): the sequence's words in timeline
# order, text edits -> catalog command batches, filler/silence review, captions, and the transcript.* commands.
# Pure C++ (model + command + ai). Owner: Local-AI engineer.
find_package(nlohmann_json CONFIG REQUIRED)

# Filler lexicons are data (data/fillers.<language>.txt); they are compiled in as bytes, so adding a language is
# adding a file and re-running CMake.
file(GLOB WABCUT_FILLER_LEXICONS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/data/fillers.*.txt)
list(SORT WABCUT_FILLER_LEXICONS)
set(_lexicon_arrays "")
set(_lexicon_entries "")
set(_lexicon_index 0)
foreach(_file IN LISTS WABCUT_FILLER_LEXICONS)
    get_filename_component(_name ${_file} NAME)
    string(REGEX REPLACE "^fillers\\.(.+)\\.txt$" "\\1" _language ${_name})
    file(READ ${_file} _hex HEX)
    string(LENGTH "${_hex}" _hex_length)
    math(EXPR _bytes "${_hex_length} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _hex "${_hex}")
    string(APPEND _lexicon_arrays "const unsigned char kLexicon${_lexicon_index}[] = {\n    ${_hex}0};\n")
    string(APPEND _lexicon_entries
           "        {std::string_view(\"${_language}\"), bytes(kLexicon${_lexicon_index}, ${_bytes})},\n")
    math(EXPR _lexicon_index "${_lexicon_index} + 1")
endforeach()
if(_lexicon_index EQUAL 0)
    message(FATAL_ERROR "wabcut_ai_edit: no filler lexicons in ${CMAKE_CURRENT_SOURCE_DIR}/data")
endif()
set(WABCUT_LEXICONS_CPP ${CMAKE_CURRENT_BINARY_DIR}/EmbeddedLexicons.cpp)
file(CONFIGURE OUTPUT ${WABCUT_LEXICONS_CPP} CONTENT
"// Generated by src/ai/edit/CMakeLists.txt from data/fillers.*.txt. Do not edit.
#include <cstddef>
#include <span>
#include <string_view>
#include <utility>

namespace wabcut::ai::edit::detail {
namespace {
${_lexicon_arrays}
std::string_view bytes(const unsigned char* data, std::size_t size) {
    return {reinterpret_cast<const char*>(data), size};
}
} // namespace

std::span<const std::pair<std::string_view, std::string_view>> embeddedLexicons() {
    static const std::pair<std::string_view, std::string_view> lexicons[] = {
${_lexicon_entries}    };
    return lexicons;
}

} // namespace wabcut::ai::edit::detail
")

add_library(wabcut_ai_edit STATIC
    src/CaptionBuilder.cpp
    src/Review.cpp
    src/SequenceTranscript.cpp
    src/TextEdits.cpp
    src/TranscriptCommands.cpp
    src/TranscriptEditing.cpp
    src/TranscriptSources.cpp
    ${WABCUT_LEXICONS_CPP})
target_include_directories(wabcut_ai_edit PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(wabcut_ai_edit PUBLIC wabcut_ai wabcut_ai_analysis wabcut_ai_search wabcut_ai_reframe wabcut_command wabcut_model
    wabcut_audio_edit wabcut_color wabcut_masks PRIVATE nlohmann_json::nlohmann_json)
wabcut_target_defaults(wabcut_ai_edit)
# editingCommands() registers the AI-9 shorts commands (src/ai/shorts), which in turn run catalog commands: the two
# static libraries need each other. wabcut_ai_shorts is defined after this directory, so link it by name.
target_link_libraries(wabcut_ai_edit PUBLIC wabcut_ai_shorts wabcut_ai_generation)
# AI-11: editingCommands() registers roughcut.* / takes.* (src/ai/roughcut) and autozoom.apply (src/ai/autozoom).
target_link_libraries(wabcut_ai_edit PUBLIC wabcut_ai_roughcut wabcut_ai_autozoom)
# AI-12: captions.translate (src/ai/translate), dub.create / narration.create (src/ai/dub).
target_link_libraries(wabcut_ai_edit PUBLIC wabcut_ai_translate wabcut_ai_dub)
# AI-13: analysis.speakers, multicam.autoSwitch (src/ai/multicam), workflow.podcast (src/ai/workflow).
target_link_libraries(wabcut_ai_edit PUBLIC wabcut_ai_multicam wabcut_ai_workflow)
# AI-14: mask.objectSelect / mask.refine / mask.propagate (src/ai/segment).
target_link_libraries(wabcut_ai_edit PUBLIC wabcut_ai_segment)
# G.2: highlights.find / highlights.toReel / highlights.toShort (src/ai/highlights).
target_link_libraries(wabcut_ai_edit PUBLIC wabcut_ai_highlights)
